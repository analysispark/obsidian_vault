```python
from pathlib import Path
import pandas as pd

root = Path(".")

files = sorted(root.rglob("*.xlsx"))

print(f"총 {len(files)}개 파일 발견\n")

base_cols = None

for file in files:
    df = pd.read_excel(file)

    cols = list(df.columns)

    if base_cols is None:
        base_cols = cols
        print(f"[기준파일] {file.name}")
        continue

    if cols == base_cols:
        print(f"[일치] {file.name}")

    else:
        print(f"\n[불일치] {file.name}")

        missing = set(base_cols) - set(cols)
        extra = set(cols) - set(base_cols)

        if missing:
            print("  누락 컬럼:")
            print("   ", sorted(missing))

        if extra:
            print("  추가 컬럼:")
            print("   ", sorted(extra))
```

```python
from pathlib import Path
import pandas as pd

# 현재 폴더 기준
root = Path(".")

# 제외할 컬럼
drop_cols = [
    "참가대회수",
    *[f"참가대회 {i}" for i in range(1, 21)]
]

dfs = []

# 하위 폴더 포함 모든 xlsx 파일 검색
for file in sorted(root.rglob("*.xlsx")):

    print(f"처리 중: {file}")

    df = pd.read_excel(file)

    # 존재하는 컬럼만 제거
    remove_cols = [col for col in drop_cols if col in df.columns]
    df = df.drop(columns=remove_cols)

    # 시즌, 대회명 추가
    df["시즌"] = file.parent.name
    df["대회명"] = file.stem

    dfs.append(df)

# 병합
merged = pd.concat(
    dfs,
    ignore_index=True,
    join="outer"
)

# 저장
output_file = "남고부_통합데이터.xlsx"
merged.to_excel(output_file, index=False)

print("\n병합 완료")
print(f"데이터 크기: {merged.shape[0]:,} 행 × {merged.shape[1]:,} 열")
print(f"저장 파일: {output_file}")
```

## 03_notebook.py

```python
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

df = pd.read_excel("남고부_통합데이터.xlsx")
print(df.head(5))


# %%

print(df.describe())

# %%
plt.rcParams["font.family"] = "AppleGothic"
plt.rcParams["axes.unicode_minus"] = False

num_df = df.select_dtypes(include="number")

# Long format으로 변환
long_df = num_df.melt(
    var_name="변인",
    value_name="값"
)

# FacetGrid
g = sns.FacetGrid(
    long_df,
    col="변인",
    col_wrap=4,      # 한 행에 4개
    sharex=False,
    sharey=False,
    height=3
)

g.map_dataframe(
    sns.histplot,
    x="값",
    bins=20
)

g.set_titles("{col_name}")

plt.tight_layout()
plt.show()
```
