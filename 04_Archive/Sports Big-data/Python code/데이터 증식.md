import cv2
import numpy as np

def video_augmentation(input_path, output_path, num_augmentations):
    # 동영상을 읽어옴
    cap = cv2.VideoCapture(input_path)
    
    # 동영상의 속성을 가져옴
    frame_width = int(cap.get(3))
    frame_height = int(cap.get(4))
    fps = cap.get(5)
    
    # 동영상 저장을 위한 VideoWriter 객체 생성
    fourcc = cv2.VideoWriter_fourcc(*'mp4v')
    out = cv2.VideoWriter(output_path, fourcc, fps, (frame_width, frame_height))
    
    # 주어진 횟수만큼 동영상을 읽어오고 증식
    for _ in range(num_augmentations):
        ret, frame = cap.read()
        
        # 이미지를 수평으로 뒤집기
        if np.random.rand() > 0.5:
            frame = cv2.flip(frame, 1)
        
        # 이미지를 수직으로 뒤집기
        if np.random.rand() > 0.5:
            frame = cv2.flip(frame, 0)
        
        # 이미지를 회전시키기
        angle = np.random.uniform(-10, 10)
        M = cv2.getRotationMatrix2D((frame_width / 2, frame_height / 2), angle, 1)
        frame = cv2.warpAffine(frame, M, (frame_width, frame_height))
        
        # 결과 동영상에 프레임 쓰기
        out.write(frame)

    # 사용한 자원 해제
    cap.release()
    out.release()
    cv2.destroyAllWindows()

# 사용 예시
video_augmentation('input_video.mp4', 'augmented_video.mp4', 5)
