import os   
from pathlib import Path  
  
parents_dir = Path(os.getcwd()).parents[1]  
  
current_dir = os.path.join(parents_dir, "Archery_DATA/2024")  
output_dir = os.path.join(current_dir, "output_excel")  
os.makedirs(output_dir, exist_ok=True)   
  
#------------------------------------------------------------------  
import pandas as pd  
  
x2_list = pd.read_excel(os.path.join(parents_dir, "Archery_DATA/2024/2배속.xlsx"), header=0)  
x2_list['DATE']= x2_list['DATE'].fillna(method='ffill', axis=0)  
x2_list = x2_list.dropna(subset=['Hardware'])  
  
#------------------------------------------------------------------  
base_path = os.path.join(Path(parents_dir.parents[1]), "DATA/Archery_2th/shoot")  
x2_list['path'] = x2_list.apply(  
    lambda x: str(Path(base_path) / str(x['DATE']) / f"{str(x['Hardware'])}600000000000000000000"),   
    axis=1)

rom collections import defaultdict  
  
def count_json_files(base_dir, exclude_paths):  
    # ... existing code ...  
      
    # 각 폴더별 json 파일 개수를 저장할 딕셔너리  
    folder_counts = defaultdict(int)  
    total_count = 0  
    # 접두사별 카운트를 저장할 딕셔너리 추가  
    prefix_counts = defaultdict(int)  
      
    for root, dirs, files in os.walk(base_dir):  
        current_path = str(Path(root).resolve())  
          
        if any(current_path.startswith(ex_path) for ex_path in exclude_paths):  
            continue  
              
        # json 파일들만 필터링  
        json_files = [f for f in files if f.endswith('.json')]  
        json_count = len(json_files)  
          
        if json_count > 0:  
            folder_counts[current_path] = json_count  
            total_count += json_count  
              
            # 각 json 파일의 접두사 카운트  
            for file in json_files:  
                if file[0].isdigit():  
                    prefix = file[:2]  
                    prefix_counts[prefix] += 1  
      
    return folder_counts, total_count, prefix_counts  
  
def count_x2_json_files(paths):  
    folder_counts = defaultdict(int)  
    total_count = 0  
    # x2 경로의 접두사별 카운트를 저장할 딕셔너리 추가  
    prefix_counts = defaultdict(int)  
      
    for path in paths:  
        if os.path.exists(path):  
            for root, _, files in os.walk(path):  
                # json 파일들만 필터링  
                json_files = [f for f in files if f.endswith('.json')]  
                json_count = len(json_files)  
                  
                if json_count > 0:  
                    folder_counts[root] = json_count  
                    total_count += json_count  
                      
                    # 각 json 파일의 접두사 카운트  
                    for file in json_files:  
                        if file[0].isdigit():  
                            prefix = file[:2]  
                            prefix_counts[prefix] += 1  
      
    return folder_counts, total_count, prefix_counts  
  
# 일반 경로 JSON 파일 개수 세기  
exclude_paths = x2_list['path'].tolist()  
folder_counts, total_count, prefix_counts = count_json_files(base_path, exclude_paths)  
  
# 2배속 경로 JSON 파일 개수 세기  
x2_counts, x2_total, x2_prefix_counts = count_x2_json_files(exclude_paths)  
  
# 기존 DataFrame 생성  
normal_df = pd.DataFrame([  
    {'Path': path, 'JSON_Count': count}   
    for path, count in folder_counts.items()  
])  
normal_df.loc[len(normal_df)] = ['Total', total_count]  
  
# prefix_counts DataFrame 생성  
prefix_df = pd.DataFrame([  
    {'Prefix': prefix, 'Count': count}   
    for prefix, count in prefix_counts.items()  
]).sort_values('Prefix')  # 접두사 순으로 정렬  
prefix_df.loc[len(prefix_df)] = ['Total', prefix_df['Count'].sum()]  
  
# 2배속 DataFrame 생성  
x2_df = pd.DataFrame([  
    {'Path': path, 'JSON_Count': count}   
    for path, count in x2_counts.items()  
])  
x2_df.loc[len(x2_df)] = ['Total', x2_total]  
  
# x2 prefix_counts DataFrame 생성  
x2_prefix_df = pd.DataFrame([  
    {'Prefix': prefix, 'Count': count}   
    for prefix, count in x2_prefix_counts.items()  
]).sort_values('Prefix')  
x2_prefix_df.loc[len(x2_prefix_df)] = ['Total', x2_prefix_df['Count'].sum()]  
  
# Excel 파일로 저장  
with pd.ExcelWriter(os.path.join(output_dir, 'json_counts_summary.xlsx')) as writer:  
    normal_df.to_excel(writer, sheet_name='Normal_Paths', index=False)  
    prefix_df.to_excel(writer, sheet_name='Normal_Prefix_Counts', index=False)  
    x2_df.to_excel(writer, sheet_name='X2_Paths', index=False)  
    x2_prefix_df.to_excel(writer, sheet_name='X2_Prefix_Counts', index=False)
